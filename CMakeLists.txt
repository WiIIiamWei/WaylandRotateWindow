cmake_minimum_required(VERSION 3.20)
project(WaylandRotateWindow VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets OpenGL DBus)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3>=0.3.65)
pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)

add_library(rotate-core STATIC src/portal.cpp src/capture.cpp src/view.cpp)
target_include_directories(rotate-core PUBLIC src)
target_link_libraries(rotate-core PUBLIC Qt6::Widgets Qt6::OpenGL Qt6::DBus
    PkgConfig::PIPEWIRE PkgConfig::EGL PkgConfig::DRM)
target_compile_options(rotate-core PRIVATE -Wall -Wextra)
add_executable(wayland-rotate-window src/main.cpp)
target_link_libraries(wayland-rotate-window PRIVATE rotate-core)

option(BUILD_TESTING "Build tests" OFF)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(test-core tests/test_core.cpp)
    target_link_libraries(test-core PRIVATE rotate-core Qt6::Test)
    add_test(NAME core COMMAND test-core)
    add_executable(test-portal tests/test_portal.cpp)
    target_link_libraries(test-portal PRIVATE rotate-core Qt6::Test)
    add_test(NAME portal COMMAND dbus-run-session -- $<TARGET_FILE:test-portal>)
    add_executable(test-render tests/test_render.cpp)
    target_link_libraries(test-render PRIVATE rotate-core Qt6::Test)
endif()

install(TARGETS wayland-rotate-window RUNTIME DESTINATION bin)
install(FILES data/io.github.WiIIiamWei.WaylandRotateWindow.desktop DESTINATION share/applications)
install(FILES LICENSE DESTINATION share/licenses/wayland-rotate-window)
